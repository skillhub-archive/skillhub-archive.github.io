# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

A. How old is a file? Dates you can count with

A real support job: logs pile up. A web shop writes a new log every day, and after a year the logs folder holds hundreds of files, most of which nobody will read again. The usual answer is to keep the recent ones where they are and move the rest into an archive folder, where they are kept but out of the way. To do that, a script has to know how old each file is.

Every file has a LastWriteTime, the moment it was last changed (Session 2 met it in Get-Member). It is a [datetime], the date-and-time type from Session 9. The lab gives every file a fixed one, so the dates below are the same on every computer:

Counting back from a date. A [datetime] carries methods, actions you call with a dot and parentheses, as you called .GetType() in Session 5. .AddDays(N) gives the date N days later, and a minus number goes back. Store the date you count from in a variable first, written as a cast: [datetime]'2026-03-04' turns the text into a date the way Session 4’s [int] turned text into a number, and it reads the text the same way on every computer. Get-Date -Date '2026-03-04', which Session 9’s tests use, reads it in your computer’s own calendar instead: with Thai settings it gives the year 1483, with Persian settings 2647, and with Saudi Arabic settings an error (all checked by switching the language for one window). A [datetime] parameter in your own function or script, such as the -AsOf you will meet in section B, reads text the way the cast does; the date parameters of PowerShell’s own commands, Get-Date -Date among them, read it in your calendar.

Why a fixed date, and not today? A job that runs every night counts from now, (Get-Date). But “now” is different every time you run it, so its answer changes with the calendar: today, every lab log is months old. A test, a drill or a page like this one needs the same answer every time, so it counts from a date it is given. That is Session 9’s -Time idea again: the date is a parameter, with now as the default. The trap: a copy can reset the dates. A job that works by age is only as good as the dates it reads. PowerShell’s Copy-Item keeps a file’s LastWriteTime, and so does Move-Item. But a copy made some other way may stamp every file with the moment of the copy, and then a whole folder of old logs looks brand new, and nothing is ever archived. Check the dates on a copy before you trust a job that reads them.

PS C:\Users\you\ps-lab> (Get-Item out/copy.log).LastWriteTime

In BashLinux’s cp stamps a copy with the time of the copy unless you add -p (“preserve”), so cp -r logs backup makes every log look new, and cp -rp logs backup keeps the dates (so does cp -a; checked on Ubuntu 26.04). The usual tool for “older than” is find FOLDER -mtime +N, counting from now. It counts a file’s age in whole days, rounded down, and lists it when that is more than N, so -mtime +1 means two days old or more.

# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

function Move-OldLog {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

function Move-OldLog {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Remove-OldReport {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { return }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  $old | Remove-Item
  $old
}

G. Where this track stops, and what comes next

Ten sessions take you from opening a terminal to a tested, scheduled module. That is real, everyday automation. It is also a fraction of PowerShell, and it is worth knowing what was left out on purpose, so you know where to look when a job needs it:

  • Remoting: running commands on other computers (Invoke-Command, Enter-PSSession). Every session here ran on your own computer.
  • Desired State Configuration (DSC): describing how a server should be set up, and letting PowerShell keep it that way.
  • Azure, Microsoft Graph and Active Directory: the large modules for cloud, Microsoft 365 and Windows domain work. They are built on everything here (cmdlets, objects, the pipeline), but each needs an account or a server this track could not give you.
  • Graphical tools and classes: windows and forms, and PowerShell’s own object types (class). Most automation needs neither.
  • Packaging a module: a module manifest (a .psd1 file with a version number), publishing it, Update-Module, and running the tests automatically on every change, which teams call CI (continuous integration). Today’s test gate is the same idea, run by your own scheduler.

The natural next step is the shell PowerShell has been compared with all along: Bash, on Linux. Every “In Bash” box in this track was a bridge to it. The Hub’s closing section lists where to go from here, including the two Microsoft certifications where PowerShell counts.

# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

function Move-OldLog {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Remove-OldReport {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { return }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  $old | Remove-Item
  $old
}

Export-ModuleMember -Function Format-LogLine, Write-Log, Get-ErrorSummary,
  Move-OldLog, Remove-OldReport
