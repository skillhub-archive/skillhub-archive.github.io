# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

function Move-OldLog {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Remove-OldReport {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { return }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  $old | Remove-Item
  $old
}

Export-ModuleMember -Function Format-LogLine, Write-Log, Get-ErrorSummary,
  Move-OldLog, Remove-OldReport

A. Your first web call: Invoke-RestMethod

A web service, often called an API, is a program on another computer that answers requests sent over the web. You send a request to an address, a URL such as https://postman-echo.com/get, and it sends back a response: usually JSON, the text format for data from Session 4. Support work is full of them: a ticket system, a shop’s order service, a supplier’s price list. If you took the APIs & Integrations track, you sent requests like these from Postman; today PowerShell sends them.

The end of a URL can carry a query: a ?, then name=value pairs joined by &. It is how a request asks for something in particular. postman-echo’s /get address answers any request by sending back what it received, so it is the safest place to start: whatever you ask, it tells you what it heard.

Invoke-RestMethod URL sends the request (a GET, the kind that asks for something) and turns the JSON reply into objects, the way Import-Csv turned a CSV’s rows into objects in Session 4. So you read the reply with the dot, as you read any property. Put the URL in single quotes: & means something else to PowerShell outside them.

a request with a query, and the reply as an object · captured live
PS C:\Users\you\ps-lab> $reply = Invoke-RestMethod                           one line, wrapped here
  'https://postman-echo.com/get?shop=fernhill&item=wok'
PS C:\Users\you\ps-lab> $reply.args

shop     item
----     ----
fernhill wok

PS C:\Users\you\ps-lab> $reply.args.item
wok
PS C:\Users\you\ps-lab> $reply.url
https://postman-echo.com/get?shop=fernhill&item=wok

args is postman-echo’s name for the query it received, and it came back as an object with one property for each pair. url is the address it was asked.

The longer way, which shows what Invoke-RestMethod saves you. Invoke-WebRequest URL sends the same request but hands you the whole response: its status code, a number saying how the request went (200 means OK; section D covers the rest), and its Content, the reply as plain text. Text is not objects, so you would turn it into objects yourself with ConvertFrom-Json, the reverse of Session 4’s ConvertTo-Json:

the same request, the long way · captured live
PS C:\Users\you\ps-lab> $page = Invoke-WebRequest                            one line, wrapped here
  'https://postman-echo.com/get?shop=fernhill' -UseBasicParsing
PS C:\Users\you\ps-lab> $page.StatusCode
200
PS C:\Users\you\ps-lab> $page.Content.GetType().Name
String
PS C:\Users\you\ps-lab> ($page.Content | ConvertFrom-Json).args.shop
fernhill

Invoke-RestMethod does all three steps in one, and when the status is not a success, one of the 200s (section D), it raises an error instead (unless it is a redirect it can follow to another address), which section D turns into something useful. Use Invoke-WebRequest when you need the status code or the raw text of a reply that worked. -UseBasicParsing changes nothing in PowerShell 7; Windows PowerShell 5.1 needs it (see the 5.1 box below).

Headers. Besides the URL, every request carries headers: named details about the request, such as which program sent it. -Headers adds your own, given as a hashtable (Session 4’s @{ name = value }). postman-echo’s /headers address sends back every header it received, under its own names in small letters:

a header of your own, and the one PowerShell sends · captured live
PS C:\Users\you\ps-lab> $heard = Invoke-RestMethod                           one line, wrapped here
  https://postman-echo.com/headers -Headers @{ 'X-Shop' = 'fernhill' }
PS C:\Users\you\ps-lab> $heard.headers.'x-shop'
fernhill
PS C:\Users\you\ps-lab> $heard.headers.'user-agent'
Mozilla/5.0 (Windows NT 10.0; Microsoft Windows 10.0.26200; en-US) PowerShell/7.6.7  yours differs

A property whose name holds a - goes in quotes after the dot. The user agent is the header in which a program names itself: PowerShell tells every server its version and your operating system. Some real services ask you to send a user agent of your own, and their documentation says what to send; Invoke-RestMethod takes -UserAgent TEXT for that.

In Bashcurl -s URL sends the same GET and prints the reply as plain text: JSON, not objects, the way Invoke-WebRequest’s Content is. Reading one value out of it takes another tool, such as jq, which Ubuntu does not install by default. That gap is the object pipeline from Session 2, now over the web.
Windows PowerShell 5.1Invoke-RestMethod works the same. Invoke-WebRequest is different without -UseBasicParsing: 5.1 reads a reply as a web page, which can run code inside it, so since a December 2025 security update it first asks a “Security Warning: Script Execution Risk” question (Microsoft’s 5.1 page), ending Do you want to continue? with No as the default. Answered N in a real 5.1 window on this workbook’s computer, it stopped with Operation cancelled due to security concerns. Use -UseBasicParsing parameter for safe HTML parsing without script execution. Run where nothing can answer it, as when a script runs on a schedule, it failed on this workbook’s computer with Object reference not set to an instance of an object. With -UseBasicParsing there is no question and it works, which is why this page always adds it (in PowerShell 7 it does nothing: 7 always reads replies the basic way). The user agent names Windows PowerShell 5.1 instead.

B. The supplier’s catalogue: a reply to filter

Now a real job. Fernhill buys its kitchenware from a supplier, and the supplier’s catalogue is a web service. dummyjson’s kitchen-accessories category plays it: about thirty products, each with a stock number. Someone asks you every Monday which products are running low. That is a question a script can answer.

The catalogue takes a query too. dummyjson’s select asks for only the fields you name. Rather than typing the query into the URL, give it to -Body as a hashtable: on a GET, Invoke-RestMethod turns each pair into the ?name=value part itself, and writes any comma or space in the safe form a URL needs.

Trap one: a ? straight after a variable. Building the URL yourself in double quotes looks natural, but PowerShell reads a ? straight after a variable’s name as part of the name. $slug?select is a variable nobody set, so it is empty, and the query vanishes with the category:

D. When the server says no: status codes and errors

Every response carries a status code. The ones you will meet most: 200 OK; 401 Unauthorized, the service did not accept who you said you were (a missing or wrong key); 404 Not Found, the address names nothing; 429 Too Many Requests, you asked too often and must wait; 500 Internal Server Error and 503 Service Unavailable, the trouble is on the server’s side. F. Get-LowStock, tests first, with Mock

Now the tool: Get-LowStock -Below N -Category NAME in the Fernhill module, giving the products with fewer than N in stock, lowest first. Sessions 9 and 10 say the tests come first. But a test that calls the real catalogue is a bad test: it needs the network, it is slow, and its answer changes whenever the supplier’s stock does. A test must give the same answer every time.

Session 10 had the answer for dates: pass the date in. A web reply cannot be passed in that way, because the function asks for it itself. This is where Mock comes in. Mock COMMAND { CODE }, inside a test, stands in for COMMAND: while the test runs, every call to COMMAND runs CODE instead, and gets CODE’s output. Mock Invoke-RestMethod, and the function receives a reply you wrote, with no network at all. Two more pieces make the fake reply:

  • [pscustomobject]@{ NAME = VALUE; NAME = VALUE }, with as many pairs as you need, turns a hashtable into an object with those properties, the same kind of object Import-Csv gives a row (Session 4) and Invoke-RestMethod gives a reply.
  • @( ITEMS ) holds a list of ITEMS, here one item per line, the way the reply’s products does.

The fake reply copies the shape of the real one from section B: a total, and a products list of objects with id, title and stock. Three more new words:

  • -ModuleName Fernhill: Get-LowStock lives in the module, and a Mock reaches a module’s own calls only when you name the module. Session 10 said so, and below you see what happens without it.
  • Should -Invoke COMMAND -Times N -Exactly checks how many times the mocked command was called during the test: here, once.
  • -ParameterFilter { TEST } counts only the calls whose parameters pass TEST. In it, each parameter is a variable: $Uri is the address Invoke-RestMethod was given. So the second test checks the function asked for the right category.

# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

function Move-OldLog {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Remove-OldReport {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { return }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  $old | Remove-Item
  $old
}

function Get-LowStock {
  [CmdletBinding()]
  param(
    [int]$Below = 15,
    [string]$Category = 'kitchen-accessories'
  )
  $url = "https://dummyjson.com/products/category/$Category"
  $reply = Invoke-RestMethod $url -Body @{ select = 'title,stock' }
  if ($reply.total -eq 0) { throw "No products in category: $Category" }
  $reply.products | Where-Object stock -lt $Below | Sort-Object stock, id
}

# Fernhill.psm1: the Fernhill Goods log tools, for any script that imports them.
$ErrorActionPreference = 'Stop'

function Format-LogLine {
  param(
    [Parameter(Mandatory)][string]$Message,
    [datetime]$Time = (Get-Date)
  )
  "$(Get-Date -Date $Time -Format 'yyyy-MM-dd HH:mm:ss') $Message"
}

function Write-Log {
  param(
    [Parameter(Mandatory)][string]$Path,
    [Parameter(Mandatory)][string]$Message
  )
  $folder = Split-Path $Path
  if ($folder -ne '') { New-Item -ItemType Directory $folder -Force | Out-Null }
  Format-LogLine -Message $Message | Add-Content $Path
}

function Get-ErrorSummary {
  param([Parameter(Mandatory)][string]$Path)
  if (-not (Test-Path $Path)) { throw "No folder to search: $Path" }
  Get-ChildItem $Path -Recurse -File | Select-String ERROR |
    Group-Object Filename | Sort-Object Count, Name -Descending |
    Select-Object @{ Name = 'Log'; Expression = { $_.Name } }, Count
}

function Get-OldFile {
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days,
    [datetime]$AsOf
  )
  $cutoff = $AsOf.AddDays(-$Days)
  Get-ChildItem $Path -File | Where-Object LastWriteTime -lt $cutoff
}

  if (-not (Test-Path $Path)) { throw "No folder to tidy: $Path" }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  New-Item -ItemType Directory "$Path/archive" -Force | Out-Null
  $old | Move-Item -Destination "$Path/archive"
  $old
}

function Remove-OldReport {
  [CmdletBinding(SupportsShouldProcess)]
  param(
    [Parameter(Mandatory)][string]$Path,
    [int]$Days = 30,
    [datetime]$AsOf = (Get-Date)
  )
  if (-not (Test-Path $Path)) { return }
  $old = Get-OldFile -Path $Path -Days $Days -AsOf $AsOf
  $old | Remove-Item
  $old
}

function Get-LowStock {
  [CmdletBinding()]
  param(
    [int]$Below = 15,
    [string]$Category = 'kitchen-accessories',
    [string]$BaseUrl = 'https://dummyjson.com',
    [string]$LogPath = "$HOME/ps-scripts/stock.log"
  )
  try {
    $url = "$BaseUrl/products/category/$Category"
    $reply = Invoke-RestMethod $url -Body @{ select = 'title,stock' }
  } catch {
    $status = [int]$_.Exception.Response.StatusCode
    Write-Log -Path $LogPath -Message "FAILED: status $status, $($_.Exception.Message)"
    throw
  }
  if ($reply.total -eq 0) {
    Write-Log -Path $LogPath -Message "FAILED: no products in category: $Category"
    throw "No products in category: $Category"
  }
  $reply.products | Where-Object stock -lt $Below | Sort-Object stock, id
}
